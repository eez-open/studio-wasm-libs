cmake_minimum_required(VERSION 3.16)
project(eez_lvgl_native_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL_VERSION "8.4.0" CACHE STRING "LVGL version to test")
set(LVGL_SOURCE "${REPO}/lvgl-runtime/v${LVGL_VERSION}/lvgl")
file(READ "${REPO}/lvgl-runtime/v${LVGL_VERSION}/lv_conf.h" CONFIG)
string(REGEX REPLACE "#define LV_USE_FREETYPE +1" "#define LV_USE_FREETYPE 0" CONFIG "${CONFIG}")
string(REGEX REPLACE "#define LV_TICK_CUSTOM +1" "#define LV_TICK_CUSTOM 0" CONFIG "${CONFIG}")
string(REPLACE "extern void emscripten_force_exit(int); emscripten_force_exit(1);" "abort();" CONFIG "${CONFIG}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h"
    "#ifndef __ASSEMBLER__\n#include <stdlib.h>\n#endif\n${CONFIG}")
set(LV_CONF_PATH "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" CACHE PATH "" FORCE)
include_directories("${CMAKE_CURRENT_BINARY_DIR}" "${LVGL_SOURCE}" "${REPO}/lvgl-runtime/v${LVGL_VERSION}")
add_compile_definitions(EEZ_FOR_LVGL LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
if(LVGL_VERSION VERSION_LESS "9.3.0")
    add_compile_definitions(LV_CONF_PATH=${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h)
endif()
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
add_link_options(-fsanitize=address,undefined)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
add_executable(lvgl-native-tests main.cpp ../lvgl-actions.cpp ../lvgl-button-matrix-memory.c
    "${REPO}/release/eez-framework-amalgamation/eez-flow.cpp"
    "${REPO}/release/eez-framework-amalgamation/eez-flow-lz4.c"
    "${REPO}/release/eez-framework-amalgamation/eez-flow-sha256.c")
target_compile_definitions(lvgl-native-tests PRIVATE EEZ_TEST_AMALGAMATION)
target_include_directories(lvgl-native-tests PRIVATE "${REPO}/release/eez-framework-amalgamation")
target_link_libraries(lvgl-native-tests lvgl m)
